# Code patches for Donkey Kong Bananza 1.0.0 main (recomp/patches.py). Each original word is checked against main.

set hidproxy DKB_SKIP_HID_PROXY 1

# al::GameFrameworkOunce::createControllerMgr (main+0x73b300) reaches sub_71008725E0 through main+0x73be90: it makes
# a transfer memory and talks to IHidServer itself (nn::hid::server::CreateHidServerProxy), past the nn::hid calls the
# host takes over, and aborts on a failed Result. The host serves no "hid" IPC service, so the call reports success.
site hidproxy main 73bea0 9404d9d0 2a1f03e0   # bl sub_71008725E0 -> mov w0, wzr

# Optional Bananza virtual display and camera aspect. Values are supplied by
# runtime/dkb_wide.c at page +0x100 (width), +0x104 (height), +0x108 (aspect),
# +0x10c (aspect / 16:9).
set dkbwide DKB_ULTRAWIDE 0

# al::getVirtualDisplayWidth / Height. The retail base uses 2880 by 1620.
site dkbwide main 5ad9f0 52816800 @page:x0 b9410000
site dkbwide main 5ada00 5280ca80 @page:x0 b9410400

# al::CameraPoseUpdater constructor, the default perspective aspect at +0x100.
# The following MOVK preserves the original upper packed FOV word.
site dkbwide main 522e44 f2a7fc68 @page:x8 b9410908

# Scene camera initialization replaces the constructor aspect through this setter.
# Scale the supplied aspect so the camera keeps its own framing and split views.
# s8 is restored immediately afterwards; s16 and x8 are caller-saved scratch.
site dkbwide main 51e0c4 bd010008 @page:x8 bd410d10 1e300910 bd010010

# Widen fresh UI projections before DrawInfo::SetProjectionMtx copies them.
# The first matrix row maps authored layout x to clip space. Saved projections
# are restored through the same setter, so changing the setter would repeatedly
# narrow the HUD. These four sites copy a newly computed sead projection into
# temporary storage instead. Keep the other matrix rows and all scratch intact.
# main+ea8a0 also projects local layout captures (dialogue backgrounds, text
# effects and icon shadows). Those textures retain their authored dimensions.
# Only widen its full 1920x1080 canvas: m00 == 2/1920 and m11 == 2/1080.
# Compare both lanes, broadcast their combined mask, and BSL the widened row
# against the original. This leaves capture rows bit-identical, preserves NZCV,
# and restores x8 and q16-q19 from the 80-byte temporary stack frame.
